# 计算拉伸后晶界属性和偏聚度。
# Calculate grain boundary excess properties, according to "10.1103/PhysRevMaterials.2.093603"
# .data file is generated from other in.file
# Created by Fusheng Tan (tanfusheng@hnu.edu.cn), 2020-05-29

### Initialization 
units  metal
dimension  3
boundary  p  p  p 
atom_style  atomic
neighbor  6.0  bin    #check the bin to see if the result is convergent
neigh_modify  delay  5  check  yes

# ### relaxation setting

variable  Temp equal 300		# 弛豫温度

variable  relax_step1  equal  50000 
variable  relax_step2  equal  50000
variable  tensile_step  equal  500000
variable  MDMC_step  equal  3000000

variable   eng_rate equal 1e-3    # unit is 1/ps, resulting erate = 1E9 / s

# ### Parameters of MD/MC
variable T equal  ${Temp} 
variable Nmc equal 1000    #每隔N步进行一次MC
variable Xmc equal 1000   #每次MC交换多少原子

#---------------------- Create GB with different [hkl](mno) --------------------      
## define the GB plane [hkl](mno)
# [001]
variable h equal 0
variable k equal 0
variable l equal 1
variable m equal 5
variable n equal 1
variable o equal 0

# The number of single cell in x y z directions
variable nx equal 10
variable ny equal 20
variable nz equal 40

variable  latconst  equal 3.14	

variable  spacex equal ${latconst}*sqrt($m^2+$n^2+$o^2)
variable  spacey equal ${latconst}*sqrt(($k*$o-$l*$n)^2+($l*$m-$h*$o)^2+($h*$n-$k*$m)^2)
variable  spacez equal ${latconst}*sqrt($h^2+$k^2+$l^2)

variable N equal $h^2+$k^2+$l^2
variable CSL_xigema equal $m^2+$N*$n^2
variable misorientation equal 2*atan($N^(1/2)*($n/$m))*180/PI

variable  sizex  equal  ${nx}*${spacex}
variable  sizey  equal  ${ny}*${spacey}
variable  sizez  equal  ${nz}*${spacez}

variable GBy equal ${sizey}*0.5  # 晶界位置

#----------------------  Model  --------------------      
#lattice  bcc  ${latconst}
region  box  block  0   ${sizex}  0  ${sizey}  0  ${sizez}  units  box
create_box  2  box

mass  1  183.8400   	# W    3.14
mass  2	186.2070  	# Re   a=2.761

#mass  1  183.8400   	# W    3.1652
#mass  2	180.94788  	# Ta   3.3039
#mass  3	4.003		# He  4.242

region  upper  block  INF  INF  ${GBy}  INF  INF  INF  units  box
#lattice  bcc  ${latconst}  orient  x  3 1 0  orient  y  -1 3 0  orient  z  0 0 1  
lattice  bcc  ${latconst}  orient  x  $m $n $o  orient  y  -$n $m $o  orient  z  $h $k $l 
create_atoms  1  region  upper

region  lower  block  INF  INF  INF  ${GBy} INF  INF  units  box
#lattice  bcc  ${latconst}  orient  x   3 -1 0  orient  y  1 3 0  orient  z  0 0 1
lattice  bcc  ${latconst}  orient  x  $m -$n $o  orient  y $n $m $o   orient  z  $h $k $l   
create_atoms  1  region  lower

group  upper region upper
group  lower region lower

#Change composition
set		group  all  type/ratio  2  0.2  23333

########### Interatomic potential #################
#pair_style     meam
#pair_coeff     *  *  W-Lee2001.library.meam  W  W-Lee2001.parameter.meam  W

pair_style     eam/alloy
pair_coeff    *  *  WRe_Bonny2017.eam.alloy  W  Re

#pair_style	eam/fs
#pair_coeff	*  *  WTaHe.eam.fs  W  Ta  He

########### displace atoms or delete atoms ##############

variable atomic_distance equal sqrt(3)/2*${latconst}  # BCC晶格的最近邻原子间距
variable del_cutoff equal  0.65*${atomic_distance}

#displace_atoms  upper  move  0  0  0  units  lattice  # 固定上层晶粒，以确保晶界位置不变（“零位移”本质是固定上层晶粒的变通方案）
#delete_atoms  overlap  ${del_cutoff}  lower  upper	 # 删除组间重叠原子
delete_atoms  overlap  ${del_cutoff}  all  all	 # 删除组间重叠原子

############### Setting ##################
compute  eatom  all  pe/atom
compute  perstress  all stress/atom  thermo_temp

compute new3d all temp  # temperature considering three directions
compute new2d all temp/partial 1 0 1 #  temperature considering two directions

################ Run minimization 1 ##########################
reset_timestep  0
thermo  100
thermo_style  custom  step  pe  lx  ly  lz  press  pxx  pyy  pzz

#fix  1  all  box/relax  iso  0  vmax  0.001

#dump  1  all  custom 10000 dump.minimization1.*.lammps id type x y z c_eatom
min_style  cg
minimize  1e-18  1e-18  10000 10000
#undump  1

write_data   sigma${CSL_xigema}[$h_$k_$l]($m_$n_$o)-minimization1.data

#unfix  1

#################Rum minimization 2 ######################
#Now allow the box to expand/contract in all directions
reset_timestep  0
thermo  100
thermo_style  custom  step  pe  lx  ly  lz  press  pxx  pyy  pzz

fix  2  all  box/relax  iso  0  vmax  0.001

#dump  2  all  custom 10000 dump.minimization2.*.lammps id type x y z c_eatom
min_style  cg
minimize  1e-18  1e-18  10000  10000
#undump 2

write_data   sigma${CSL_xigema}[$h_$k_$l]($m_$n_$o)-minimization2.data

unfix  2

print  "---------- The GB sigma${CSL_xigema}[$h_$k_$l]($m_$n_$o) is build -----------"


################# Rum minimization 3 ######################
#Now allow the box to expand/contract perpendicular to the GB
reset_timestep  0
thermo  100
thermo_style  custom  step  pe  lx  ly  lz  press  pxx  pyy  pzz

fix  3  all  box/relax  y  0  vmax  0.001

#dump  3  all  custom 10000 dump.minimization3.*.lammps id type x y z c_eatom
min_style  cg
minimize  1e-18  1e-18  10000  10000
#undump 3

write_data   sigma${CSL_xigema}[$h_$k_$l]($m_$n_$o)-minimization3.data

unfix  3

print  "---------- The GB sigma${CSL_xigema}[$h_$k_$l]($m_$n_$o) is build -----------"


########################### Relaxation ################################ 
# ### Step1:  release the pressure in all direction, let the model expands 

velocity  all create ${Temp} 233333 dist gaussian units box 

reset_timestep 0

timestep 0.001
thermo  100
thermo_style custom step temp pe press pxx pyy pzz

dump  1  all  custom  10000  dump_relax_${Temp}K.*.lammpstrj  id  type x y z  c_eatom 

fix  1  all  npt  temp  ${Temp}  ${Temp}  0.1  iso 0 0 1   #Tdamp=100*dt; Pdamp=1000*dt
run  ${relax_step1}

unfix 1
undump 1

write_data  sigma${CSL_xigema}[$h_$k_$l]($m_$n_$o)-Relaxed.data

print  "---------- 全方位驰豫晶界已完成 -----------"

###Store final cell length for strain calculation
variable tmpy equal ly
variable L0 equal ${tmpy}
print  "Initial length, L0 = ${L0}"

#----------------------------- 设置一些计算 ------------------------------------
###能量计算
compute  perpe_initial all pe/atom		#计算每个原子的势能
compute  PE_initial_all all reduce sum c_perpe_initial     # 计算模型总势能


###体积计算
compute  pervolume_initial all voronoi/atom
compute  VOL_initial_all all reduce sum c_pervolume_initial[1]     # 计算模型总体积


###对称性计算
compute  percs_initial all centro/atom bcc	#compute ID group-ID centro/atom lattice keyword value 
compute  CS_initial_all all reduce sum c_percs_initial	# 计算模型总CS


###方向有序度计算
compute  perQ_initial all orientorder/atom               # 默认的晶体结构是BCC，即计算一个原子周围8个近邻原子（1NN=8）
compute  Q6_initial_all all reduce sum c_perQ_initial[2]     # 计算模型总键序有序度 Q_6


########################### 初始状态偏聚 ################################ 
reset_timestep 0

dump 	MDMC  all  custom 10000  dump_SGMC_initialModel.*.lammpstrj  id  type x y z c_perpe_initial

fix 			3 all nvt temp ${Temp} ${Temp} 0.1  # aniso 0.0 0.0 1.5
fix_modify	3 temp new2d

#如果体系初始即固定了成分，那么后续采用传统mc/md就行，单纯交换原子然后判定体系势能，方法简单，结果稳定。但是耗时长。
fix 		mySGMC12  all atom/swap ${Nmc} ${Xmc} 2233 ${Temp} semi-grand no types 1 2 ke yes

thermo  100
thermo_style custom step temp pe  lx  ly  lz  press  pxx  pyy  pzz  c_PE_initial_all  c_VOL_initial_all  c_CS_initial_all  c_Q6_initial_all

run 		${MDMC_step}

unfix 		3
unfix 		mySGMC12 
undump 	MDMC 


########################### 拉伸 ################################ 

reset_timestep 0

### In units metal, pressure is in [bars]; p2, p3, p4 are in GPa
variable      deltaL  equal  (ly-${L0})
variable      p1  equal  v_deltaL
variable      p2  equal  -pxx/10000
variable      p3  equal  -pyy/10000
variable      p4  equal  -pzz/10000
variable      strainy equal v_deltaL/${L0}
fix		 def1 all print 100 "${strainy} ${p3}" file  strain_stress_tensile_data.txt screen no

###display thermo message
thermo 100
thermo_style custom step temp pe  lx  ly  lz  press  pxx  pyy  pzz v_strainy v_p2 v_p3 v_p4 

dump 1 all custom 10000 dump_tensile.*.lammps id type x y z c_eatom  & 
c_perstress[1] c_perstress[2] c_perstress[3] c_perstress[4] c_perstress[5] c_perstress[6] 

fix 1 all npt temp ${Temp} ${Temp} 0.1 x 0 0 1 z 0 0 1 drag 0.2
fix_modify	1 temp new2d

fix 2 all deform 1 y erate ${eng_rate} units box remap x

run ${tensile_step}
write_restart		model-tensile.restart
write_data		model-tensile.data

undump 1
unfix  1
unfix  2
unfix def1

print "-------------- current strain = ${strainy}-----------------"
