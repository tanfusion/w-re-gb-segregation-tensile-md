# 计算拉伸后晶界属性和偏聚度。
# Calculate grain boundary excess properties, according to "10.1103/PhysRevMaterials.2.093603"
# .data file is generated from other in.file
# Created by Fusheng Tan (tanfusheng@hnu.edu.cn), 2020-05-29

### Initialization 
units  metal
dimension  3
boundary  p  p  p 
atom_style  atomic
neighbor  4.0  bin    #check the bin to see if the result is convergent
neigh_modify  delay  5  check  yes

# ### relaxation setting

variable  Temp equal 300		# 弛豫温度

variable  latconst  equal  3.14

variable  relax_step1  equal  50000 
variable  relax_step2  equal  50000
variable  tensile_step  equal  500000
variable  compress_step  equal   2*${tensile_step}
variable  compress_tensile_step  equal  ${tensile_step}
variable  MDMC_step  equal  2000000

variable   eng_rate equal 1e-3    # unit is 1/ps, resulting erate = 1E9 / s

# ### Parameters of MD/MC
variable T equal  ${Temp} 
variable Nmc equal 1000    #每隔N步进行一次MC
variable Xmc equal 1000   #每次MC交换多少原子

read_data	W_Poly_grain25.lmp  

# ### Change composition
set		group  all  type/ratio  2  0.2  23333  #Re的原子比

########### Interatomic potential #################
pair_style     eam/alloy
pair_coeff    *  *  WRe_Bonny2017.eam.alloy  W  Re

########### displace atoms or delete atoms ##############

variable atomic_distance equal sqrt(3)/2*${latconst}  # BCC晶格的最近邻原子间距
variable del_cutoff equal  0.65*${atomic_distance}

#displace_atoms  upper  move  0  0  0  units  lattice  # 固定上层晶粒，以确保晶界位置不变（“零位移”本质是固定上层晶粒的变通方案）
#delete_atoms  overlap  ${del_cutoff}  lower  upper	 # 删除组间重叠原子
delete_atoms  overlap  ${del_cutoff}  all  all	 # 删除组间重叠原子

############### Setting ##################
compute  eatom  all  pe/atom
compute  perstress  all stress/atom  thermo_temp

compute new3d all temp  # temperature considering three directions
compute new2d all temp/partial 1 0 1 #  temperature considering two directions

################ Run minimization 1 ##########################
reset_timestep  0
thermo  100
thermo_style  custom  step  pe  lx  ly  lz  press  pxx  pyy  pzz

#fix  1  all  box/relax  iso  0  vmax  0.001

#dump  1  all  custom 10000 dump.minimization1.*.lammps id type x y z c_eatom
min_style  cg
minimize  1e-18  1e-18  10000 10000
#undump  1

write_data   Model_Minimization1.data

#unfix  1

#################Rum minimization 2 ######################
#Now allow the box to expand/contract in all directions
reset_timestep  0
thermo  100
thermo_style  custom  step  pe  lx  ly  lz  press  pxx  pyy  pzz

fix  2  all  box/relax  iso  0  vmax  0.001

#dump  2  all  custom 10000 dump.minimization2.*.lammps id type x y z c_eatom
min_style  cg
minimize  1e-18  1e-18  10000  10000
#undump 2

write_data   Model_Minimization2.data

unfix  2

################# Rum minimization 3 ######################
#Now allow the box to expand/contract perpendicular to the GB
reset_timestep  0
thermo  100
thermo_style  custom  step  pe  lx  ly  lz  press  pxx  pyy  pzz

fix  3  all  box/relax  y  0  vmax  0.001

#dump  3  all  custom 10000 dump.minimization3.*.lammps id type x y z c_eatom
min_style  cg
minimize  1e-18  1e-18  10000  10000
#undump 3

write_data   Model_Minimization3.data

unfix  3

print  "---------- The Minimization3 is done -----------"


########################### Relaxation ################################ 
# ### Step1:  release the pressure in all direction, let the model expands 

velocity  all create ${Temp} 233333 dist gaussian units box 

reset_timestep 0

timestep 0.001
thermo  100
thermo_style custom step temp pe press pxx pyy pzz

dump  1  all  custom  10000  dump_relax_${Temp}K.*.lammpstrj  id  type x y z  c_eatom 

fix  1  all  npt  temp  ${Temp}  ${Temp}  0.1  iso 0 0 1   #Tdamp=100*dt; Pdamp=1000*dt
run  ${relax_step1}

unfix 1
undump 1

write_data	Model_Relaxed.data

print  "---------- 全方位驰豫已完成 -----------"

###Store final cell length for strain calculation
variable tmpy equal ly
variable L0 equal ${tmpy}
print  "Initial length, L0 = ${L0}"


########################### 初始状态偏聚 ################################ 
reset_timestep 0

dump 	MDMC  all  custom 10000  dump_SGMC_initialModel.*.lammpstrj  id  type x y z

fix 			3 all nvt temp ${Temp} ${Temp} 0.1  # aniso 0.0 0.0 1.5
fix_modify	3 temp new2d

#如果体系初始即固定了成分，那么后续采用传统mc/md就行，单纯交换原子然后判定体系势能，方法简单，结果稳定。但是耗时长。
fix 		mySGMC12  all atom/swap ${Nmc} ${Xmc} 2233 ${Temp} semi-grand no types 1 2 ke yes

thermo  100
thermo_style custom step temp pe  lx  ly  lz  press  pxx  pyy  pzz

run 		${MDMC_step}

unfix 		3
unfix 		mySGMC12 
undump 	MDMC 

########################### 拉伸 ################################ 

reset_timestep 0

### In units metal, pressure is in [bars]; p2, p3, p4 are in GPa
variable      deltaL  equal  (ly-${L0})
variable      p1  equal  v_deltaL
variable      p2  equal  -pxx/10000
variable      p3  equal  -pyy/10000
variable      p4  equal  -pzz/10000
variable      strainy equal v_deltaL/${L0}
fix def1 all print 100 "${strainy} ${p3}" file Tensile_data.txt screen no

###display thermo message
thermo 100
thermo_style custom step temp pe  lx  ly  lz  press  pxx  pyy  pzz v_strainy v_p2 v_p3 v_p4 

dump 1 all custom 10000 dump_tensile.*.lammps id type x y z c_eatom  & 
c_perstress[1] c_perstress[2] c_perstress[3] c_perstress[4] c_perstress[5] c_perstress[6] 

fix 1 all npt temp ${Temp} ${Temp} 0.1 x 0 0 1 z 0 0 1 drag 0.2
fix_modify	1 temp new2d

fix 2 all deform 1 y erate ${eng_rate} units box remap x

run ${tensile_step}
write_restart		model-tensile.restart
write_data		model-tensile.data

undump 1
unfix  1
unfix  2

print "-------------- current strain = ${strainy}-----------------"
